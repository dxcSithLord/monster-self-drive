# MonsterBorg Self-Drive Project Requirements
# Python 3.13.5+ (Debian Trixie default)
# Target: Raspberry Pi 3B with Raspberry Pi OS Trixie / Ubuntu 24.04+

# Core Dependencies
# NOTE: opencv-python 4.10.0.84+ required for NumPy 2.x compatibility
# (4.10.0.82 and earlier are incompatible). Clear pip cache if issues arise:
#   pip cache purge

opencv-python>=4.10.0.84
opencv-contrib-python>=4.10.0.84  # For extended tracking algorithms

# Raspberry Pi Camera Support (install separately on Pi)
# This project uses picamera2 (libcamera stack). The legacy picamera (MMAL) is NOT supported.
# picamera2 has C extensions requiring build tools. Two options:
#
# Option 1 (Recommended): Use system packages (no pip install needed)
#   sudo apt install python3-picamera2
#   Then create venv with: python3 -m venv --system-site-packages monster_venv
#
# Option 2: Build from source in venv (slower, needs dev packages)
#   sudo apt install python3-dev libcap-dev libarchive-dev
#   pip install picamera2

# Alternative: For development on non-Pi systems
# opencv-python will use regular webcam

# Machine Learning - Object Detection (Phase 2)
# LiteRT (formerly TensorFlow Lite) for CPU inference
# ai-edge-litert is Google's successor to tflite-runtime
# Verified working on Python 3.13 + Debian Trixie + aarch64
# See: https://ai.google.dev/edge/litert/migration
ai-edge-litert>=2.1.0

# Google Coral TPU Support (Phase 2 - Hardware Acceleration)
# Provides ~25-30 fps object detection on Raspberry Pi
#
# IMPORTANT: pycoral is NOT available for Python 3.13 yet.
# The code uses ai-edge-litert with Edge TPU delegate as fallback.
#
# System setup for Coral TPU (Debian 13 Trixie):
#   # Install Edge TPU runtime
#   curl -fsSL https://packages.cloud.google.com/apt/doc/apt-key.gpg | \
#       sudo gpg --dearmor -o /etc/apt/keyrings/coral-edgetpu.gpg
#   echo "deb [signed-by=/etc/apt/keyrings/coral-edgetpu.gpg] https://packages.cloud.google.com/apt coral-edgetpu-stable main" | \
#       sudo tee /etc/apt/sources.list.d/coral-edgetpu.list
#   sudo apt update && sudo apt install libedgetpu1-std
#
# For Python 3.11 environments where pycoral IS available:
#   pip install --index-url https://google-coral.github.io/py-repo/ pycoral~=2.0
#
# NOTE: Do NOT use --extra-index-url for pycoral (PyPI has wrong package)
